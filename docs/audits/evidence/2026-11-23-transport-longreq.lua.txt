-- One long request on a fresh keep-alive connection to PostgREST, then time
-- the idle close measured from the END of the response.
local s = tonumber(arg[1])
local sock = ngx.socket.tcp(); sock:settimeout(130000)
assert(sock:connect("supabase_rest_cvfit_odm2", 3000))
local body = '{"s":' .. s .. '}'
ngx.update_time(); local t0 = ngx.now()
assert(sock:send("POST /rpc/p0_sleep_probe HTTP/1.1\r\nHost: rest\r\nContent-Type: application/json\r\nContent-Length: " .. #body .. "\r\nConnection: keep-alive\r\n\r\n" .. body))
local status = sock:receive("*l"); local len = 0
while true do local l = assert(sock:receive("*l")); local cl = l:match("^[Cc]ontent%-[Ll]ength:%s*(%d+)"); if cl then len = tonumber(cl) end; if l == "" then break end end
if len > 0 then sock:receive(len) end
ngx.update_time(); local t1 = ngx.now()
local _, err = sock:receive(1); ngx.update_time()
print(string.format("sleep=%ss status=[%s] response_after=%.1fs idle_close_after_response=%.1fs (%s)", s, status, t1 - t0, ngx.now() - t1, tostring(err)))
