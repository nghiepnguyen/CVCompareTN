local sock = ngx.socket.tcp(); sock:settimeout(3000)
assert(sock:connect("127.0.0.1", 8001))
sock:send("GET / HTTP/1.0\r\nHost: localhost\r\n\r\n")
local body = sock:receive("*a")
local cjson = require "cjson"
local c = cjson.decode(body:match("\r\n\r\n(.*)$")).configuration
print("idle=", c.upstream_keepalive_idle_timeout, " pool=", c.upstream_keepalive_pool_size, " maxreq=", c.upstream_keepalive_max_requests)
